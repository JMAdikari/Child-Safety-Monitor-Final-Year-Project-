python -c "
import os, csv

# Check which fall videos are in CSV and which are missing
fall_folder = 'data/raw data/fall'
csv_path = 'data/ground_truth.csv'

# Get all videos in fall folder
fall_videos = set()
if os.path.exists(fall_folder):
    for f in os.listdir(fall_folder):
        if f.lower().endswith(('.mp4','.avi','.mkv','.mov')):
            fall_videos.add(f)

# Get all videos referenced in CSV
csv_videos = set()
with open(csv_path, 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        vf = row['video_file'].strip()
        if 'fall' in vf.lower():
            csv_videos.add(os.path.basename(vf))

# Compare
print(f'Videos in fall folder: {len(fall_videos)}')
print(f'Fall entries in CSV: {len(csv_videos)}')

not_in_csv = fall_videos - csv_videos
in_csv = fall_videos & csv_videos

print(f'\nIN CSV ({len(in_csv)}):')
for v in sorted(in_csv):
    print(f'  OK: {v}')

print(f'\nNOT IN CSV ({len(not_in_csv)}):')
for v in sorted(not_in_csv):
    print(f'  MISSING: {v} — needs to be added')

if not not_in_csv:
    print('\n  All fall videos are in CSV!')
